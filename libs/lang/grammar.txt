program -> declaration* EOF;
declaration ->
    classDeclaration |
    funcDeclaration |
    letDeclaration |
    statement;
letDeclaration -> "let" IDENTIFIER ("=" expressions)? ";";
funcDeclaration -> "func" callableDeclaration;
classDeclaration -> "class" IDENTIFIER "{" classBody "}";
classBodyDeclaration -> (callableDeclaration | staticCallableDeclaration | getterDeclaration | setterDeclaration)*;
staticCallableDeclaration -> "static" callableDeclaration;
getterDeclaration -> "get" IDENTIFIER "(" ")" blockStatement;
setterDeclaration -> "set" IDENTIFIER "(" IDENTIFIER ")" blockStatement;
callableDeclaration -> IDENTIFIER parametersAndBody;
lambdaExpression -> "func" parametersAndBody;
parametersAndBody -> "(" parameters? ")" blockStatement;
parameters -> IDENTIFIER ("," IDENTIFIER)*;
statement ->
    expressionStatement |
    blockStatement |
    ifStatement |
    whileStatement |
    forStatement |
    breakStatement |
    continueStatement;
ifStatement -> "if" "(" expressions ")" statement ("else" statement)?;
whileStatement -> "while" "(" expressions ")" statement;
forStatement -> "for" "(" (letDeclaration | expressionStatement | ";") expressions? ";" expressions? ")"
    statement;
breakStatement -> "break" ";";
continueStatement -> "continue" ";";
returnStatement -> "return" expressions? ";";
expressionStatement -> expressions ";";
blockStatement -> "{" declaration* "}";
expressions -> expression ("," expression)*;
expression -> conditional;
assignment -> (newInstance ".")? IDENTIFIER "=" assignment | conditional;
conditional -> logicalAnd ("?" logicalAnd ":" conditional)*;
logicalOr -> logicalAnd ("||" logicalAnd)*;
logicalAnd -> equality ("&&" equality)*;
equality -> comparison (("=="|"!=") comparison)*;
comparison -> bitwiseOr ((">"|"<"|">="|"<=") bitwiseOr)*;
bitwiseOr -> bitwiseXor ("|" bitwiseXor)*;
bitwiseXor -> bitwiseAnd ("^" bitwiseAnd)*;
bitwiseAnd -> term ("&" term)*;
shift -> term (("<<"|">>") term)*;
term -> factor (("+"|"-") factor)*;
factor -> unary (("*"|"/") unary)*;
unary -> ("!"|"-"|"~") unary | newInstance;
newInstance -> "new" primary "(" arguments? ")" | callAndGet;
callAndGet -> primary ("(" arguments? ")" | "." IDENTIFIER)*;
arguments -> expression ("," expression)*;
primary ->
    NUMBER |
    STRING |
    "true" |
    "false" |
    "nil" |
    "(" expression ")" |
    this |
    IDENTIFIER |
    lambdaExpression;
